# Step 4: Async Processing and Scheduling

### GIMBAP Async with Goroutines

```go
// internal/async/executor.go
package async

import (
    "context"
    "sync"
    "time"
    
    "github.com/jhseong7/gimbap"
)

// TaskExecutor manages goroutines and concurrent task execution
type TaskExecutor struct {
    workerPool chan struct{}
    wg         sync.WaitGroup
    ctx        context.Context
    cancel     context.CancelFunc
}

func NewTaskExecutor(maxWorkers int) *TaskExecutor {
    ctx, cancel := context.WithCancel(context.Background())
    return &TaskExecutor{
        workerPool: make(chan struct{}, maxWorkers),
        ctx:        ctx,
        cancel:     cancel,
    }
}

func (e *TaskExecutor) Execute(task func()) {
    e.wg.Add(1)
    go func() {
        defer e.wg.Done()
        
        // Acquire worker slot
        e.workerPool <- struct{}{}
        defer func() { <-e.workerPool }()
        
        // Execute task
        task()
    }()
}

func (e *TaskExecutor) ExecuteWithContext(ctx context.Context, task func(context.Context)) {
    e.wg.Add(1)
    go func() {
        defer e.wg.Done()
        
        // Acquire worker slot
        e.workerPool <- struct{}{}
        defer func() { <-e.workerPool }()
        
        // Execute task with context
        select {
        case <-ctx.Done():
            return
        default:
            task(ctx)
        }
    }()
}

func (e *TaskExecutor) Wait() {
    e.wg.Wait()
}

func (e *TaskExecutor) Shutdown() {
    e.cancel()
    e.Wait()
}

var TaskExecutorProvider = gimbap.DefineProvider(gimbap.ProviderOption{

        s.logger.Printf("Sending email to %s", to)

        time.Sleep(2 * time.Second)

}

// Fire-and-forget async method
func (s *EmailService) SendBulkEmails(requests []EmailRequest) {
    s.executor.Execute(func() {
        s.logger.Printf("Processing %d emails", len(requests))
        
        // Use goroutines for parallel processing
        var wg sync.WaitGroup
        semaphore := make(chan struct{}, 5) // Limit to 5 concurrent sends
        
        for _, request := range requests {
            wg.Add(1)
            go func(req EmailRequest) {
                defer wg.Done()
                
                semaphore <- struct{}{}
                defer func() { <-semaphore }()
                
                if err := s.sendEmailSync(req); err != nil {
                    s.logger.Printf("Failed to send email to %s: %v", req.To, err)
                }
            }(request)
        }
        
        wg.Wait()
    })
}

// Async with transaction-like behavior
func (s *EmailService) ProcessOrderAsync(order *Order) <-chan error {
    errChan := make(chan error, 1)
    
    s.executor.Execute(func() {
        // Use a transaction-like pattern
        if err := s.processOrderTransaction(order); err != nil {
            errChan <- err
        } else {
            errChan <- nil
        }
        close(errChan)
    })
    
    return errChan
}

func (s *EmailService) processOrderTransaction(order *Order) error {
    // Simulate transaction
    tx := s.db.Begin()
    defer func() {
        if r := recover(); r != nil {
            tx.Rollback()
        }
    }()
    
    if err := s.orderRepo.SaveTx(tx, order); err != nil {
        tx.Rollback()
        return err
    }
    
    // These can be async within the transaction
    go s.notificationService.NotifyCustomer(order.CustomerID)
    go s.inventoryService.UpdateStock(order.Items)
    
    return tx.Commit()
}
```

```java
@Service
public class OrderProcessingService {
    
    @Async
    public CompletableFuture<OrderResult> processOrder(OrderRequest request) {
        return CompletableFuture
            .supplyAsync(() -> validateOrder(request))
            .thenCompose(order -> checkInventory(order))
            .thenCompose(order -> calculatePricing(order))
            .thenCompose(order -> createOrder(order))
            .thenApply(order -> new OrderResult(order.getId(), "Success"))
            .exceptionally(ex -> new OrderResult(null, ex.getMessage()));
    }
    
    public CompletableFuture<PaymentResult> processPaymentWithRetry(Payment payment) {
        return CompletableFuture
            .supplyAsync(() -> processPayment(payment))
            .thenApply(CompletableFuture::completedFuture)
            .exceptionally(ex -> {
                // Retry logic
                return CompletableFuture
                    .delayedExecutor(1, TimeUnit.SECONDS)
                    .execute(() -> processPayment(payment));
            })
            .thenCompose(Function.identity());
    }
    
    @Async
    public CompletableFuture<List<ProductInfo>> fetchProductsParallel(List<Long> productIds) {
        List<CompletableFuture<ProductInfo>> futures = productIds.stream()
            .map(id -> CompletableFuture.supplyAsync(() -> fetchProduct(id)))
            .collect(Collectors.toList());
        
        return CompletableFuture.allOf(futures.toArray(new CompletableFuture[0]))
            .thenApply(v -> futures.stream()
                .map(CompletableFuture::join)
                .collect(Collectors.toList()));
    }
}
```

### GIMBAP Channels and Goroutines

```go
// internal/service/order_processing.go
package service

import (
    "context"
    "fmt"
    "sync"
    "time"
)

type OrderProcessingService struct {
    executor     *async.TaskExecutor
    orderRepo    *repository.OrderRepository
    inventoryService *InventoryService
    pricingService  *PricingService
}

// Process order with chained operations
func (s *OrderProcessingService) ProcessOrder(ctx context.Context, request OrderRequest) <-chan OrderResult {
    resultChan := make(chan OrderResult, 1)
    
    go func() {
        defer close(resultChan)
        
        // Chain operations using channels
        orderChan := s.validateOrderAsync(ctx, request)
        
        select {
        case order := <-orderChan:
            if order.Error != nil {
                resultChan <- OrderResult{Error: order.Error}
                return
            }
            
            // Continue chain
            inventoryChan := s.checkInventoryAsync(ctx, order.Order)
            
            select {
            case invResult := <-inventoryChan:
                if invResult.Error != nil {
                    resultChan <- OrderResult{Error: invResult.Error}
                    return
                }
                
                // Continue with pricing
                pricingChan := s.calculatePricingAsync(ctx, invResult.Order)
                
                select {
                case priceResult := <-pricingChan:
                    if priceResult.Error != nil {
                        resultChan <- OrderResult{Error: priceResult.Error}
                        return
                    }
                    
                    // Create order
                    createChan := s.createOrderAsync(ctx, priceResult.Order)
                    
                    select {
                    case finalResult := <-createChan:
                        resultChan <- finalResult
                    case <-ctx.Done():
                        resultChan <- OrderResult{Error: ctx.Err()}
                    }
                    
                case <-ctx.Done():
                    resultChan <- OrderResult{Error: ctx.Err()}
                }
                
            case <-ctx.Done():
                resultChan <- OrderResult{Error: ctx.Err()}
            }
            
        case <-ctx.Done():
            resultChan <- OrderResult{Error: ctx.Err()}
        }
    }()
    
    return resultChan
}

// Alternative: Pipeline pattern
func (s *OrderProcessingService) ProcessOrderPipeline(ctx context.Context, request OrderRequest) <-chan OrderResult {
    // Create pipeline stages
    validated := s.validateOrderStage(ctx, request)
    checked := s.checkInventoryStage(ctx, validated)
    priced := s.calculatePricingStage(ctx, checked)
    created := s.createOrderStage(ctx, priced)
    
    return created
}

// Pipeline stages
func (s *OrderProcessingService) validateOrderStage(ctx context.Context, request OrderRequest) <-chan OrderData {
    out := make(chan OrderData, 1)
    
    go func() {
        defer close(out)
        
        order, err := s.validateOrder(request)
        select {
        case out <- OrderData{Order: order, Error: err}:
        case <-ctx.Done():
        }
    }()
    
    return out
}

func (s *OrderProcessingService) checkInventoryStage(ctx context.Context, in <-chan OrderData) <-chan OrderData {
    out := make(chan OrderData, 1)
    
    go func() {
        defer close(out)
        
        select {
        case data := <-in:
            if data.Error != nil {
                out <- data
                return
            }
            
            checked, err := s.checkInventory(data.Order)
            select {
            case out <- OrderData{Order: checked, Error: err}:
            case <-ctx.Done():
            }
            
        case <-ctx.Done():
        }
    }()
    
    return out
}

// Process payment with retry
func (s *OrderProcessingService) ProcessPaymentWithRetry(ctx context.Context, payment Payment) <-chan PaymentResult {
    resultChan := make(chan PaymentResult, 1)
    
    go func() {
        defer close(resultChan)
        
        maxRetries := 3
        backoff := time.Second
        
        for i := 0; i < maxRetries; i++ {
            result, err := s.processPayment(payment)
            if err == nil {
                resultChan <- result
                return
            }
            
            if i < maxRetries-1 {
                select {
                case <-time.After(backoff):
                    backoff *= 2 // Exponential backoff
                case <-ctx.Done():
                    resultChan <- PaymentResult{Error: ctx.Err()}
                    return
                }
            }
        }
        
        resultChan <- PaymentResult{Error: fmt.Errorf("payment failed after %d retries", maxRetries)}
    }()
    
    return resultChan
}

// Fetch products in parallel
func (s *OrderProcessingService) FetchProductsParallel(ctx context.Context, productIDs []int64) <-chan []ProductInfo {
    resultChan := make(chan []ProductInfo, 1)
    
    go func() {
        defer close(resultChan)
        
        var wg sync.WaitGroup
        products := make([]ProductInfo, len(productIDs))
        
        for i, id := range productIDs {
            wg.Add(1)
            go func(index int, productID int64) {
                defer wg.Done()
                
                product, err := s.fetchProduct(ctx, productID)
                if err != nil {
                    products[index] = ProductInfo{Error: err}
                } else {
                    products[index] = product
                }
            }(i, id)
        }
        
        wg.Wait()
        
        select {
        case resultChan <- products:
        case <-ctx.Done():
        }
    }()
    
    return resultChan
}

// Wait for multiple futures
func WaitAll(ctx context.Context, channels ...<-chan interface{}) <-chan []interface{} {
    resultChan := make(chan []interface{}, 1)
    
    go func() {
        defer close(resultChan)
        
        results := make([]interface{}, len(channels))
        
        for i, ch := range channels {
            select {
            case result := <-ch:
                results[i] = result
            case <-ctx.Done():
                return
            }
        }
        
        resultChan <- results
    }()
    
    return resultChan
}
```

```go
    "context"
    "log"
    "sync"
    "time"
    
    "github.com/jhseong7/gimbap"
    "github.com/jhseong7/gimbap/microservice"

)

type ScheduledTasks struct {
    microservice.IMicroService
    cron           *cron.Cron
    reportService  *ReportService
    cacheService   *CacheService
    metricsService *MetricsService
    logger         *log.Logger

    mu           sync.RWMutex

    ctx    context.Context
    cancel context.CancelFunc
}

func NewScheduledTasks(
    reportService *ReportService,
    cacheService *CacheService,
    metricsService *MetricsService,
) *ScheduledTasks {

        reportService:  reportService,
        cacheService:   cacheService,

        cancel:         cancel,

}

func (s *ScheduledTasks) Start() {
    s.logger.Println("Starting scheduled tasks")
    
    // Fixed delay - waits 5 seconds after completion
    s.scheduleWithFixedDelay(5*time.Second, s.reportCurrentTime)
    
    // Fixed rate - runs every 10 seconds
    s.scheduleAtFixedRate(10*time.Second, s.collectMetrics)
    
    // Cron expression - runs at 2 AM every day
    s.cron.AddFunc("0 0 2 * * *", s.generateDailyReport)
    
    // Cron with specific timezone
    nyLocation, _ := time.LoadLocation("America/New_York")
    s.cron.AddFunc("0 0 9-17 * * MON-FRI", s.withLocation(nyLocation, s.businessHoursTask))
    
    // Initial delay then fixed rate
    s.scheduleWithInitialDelay(60*time.Second, 5*time.Minute, s.cleanupCache)
    
    // Dynamic scheduling
    s.startDynamicScheduler()
    
    // Start cron scheduler
    s.cron.Start()
}

func (s *ScheduledTasks) Stop() {
    s.logger.Println("Stopping scheduled tasks")
    s.cancel()
    s.cron.Stop()
}

// Fixed delay implementation
func (s *ScheduledTasks) scheduleWithFixedDelay(delay time.Duration, task func()) {
    go func() {
        for {
            select {
            case <-s.ctx.Done():
                return
            default:
                task()
                time.Sleep(delay)
            }
        }
    }()
}

// Fixed rate implementation
func (s *ScheduledTasks) scheduleAtFixedRate(period time.Duration, task func()) {
    go func() {
        ticker := time.NewTicker(period)
        defer ticker.Stop()
        
        for {
            select {
            case <-s.ctx.Done():
                return
            case <-ticker.C:
                go task() // Run in separate goroutine to maintain rate
            }
        }
    }()
}

// Initial delay then fixed rate
func (s *ScheduledTasks) scheduleWithInitialDelay(initialDelay, period time.Duration, task func()) {
    go func() {
        select {
        case <-s.ctx.Done():
            return
        case <-time.After(initialDelay):
            // After initial delay, schedule at fixed rate
            s.scheduleAtFixedRate(period, task)
        }
    }()
}

type DynamicTask struct {

    CronExpr string

}

func (s *ScheduledTasks) startDynamicScheduler() {
    go func() {
        ticker := time.NewTicker(1 * time.Minute)
        defer ticker.Stop()
        
        for {
            select {
            case <-s.ctx.Done():
                return
            case <-ticker.C:

        }

func (s *ScheduledTasks) updateDynamicTasks() {
    // Get task configurations from database
    configs := s.getTaskConfigsFromDatabase()
    
    s.mu.Lock()
    defer s.mu.Unlock()
    
    // Remove old tasks
    for id, task := range s.dynamicTasks {
        if _, exists := configs[id]; !exists {
            s.cron.Remove(task.EntryID)
            delete(s.dynamicTasks, id)
            s.logger.Printf("Removed dynamic task: %s", id)
        }
    }
    
    // Add or update tasks
    for id, config := range configs {
        existing, exists := s.dynamicTasks[id]
        
        if !exists || existing.CronExpr != config.CronExpr {
            // Remove old entry if exists
            if exists {
                s.cron.Remove(existing.EntryID)
            }
            
            // Add new entry
            entryID, err := s.cron.AddFunc(config.CronExpr, config.Task)
            if err != nil {
                s.logger.Printf("Failed to add dynamic task %s: %v", id, err)
                continue
            }
            
            s.dynamicTasks[id] = &DynamicTask{
                ID:       id,
                CronExpr: config.CronExpr,
                Task:     config.Task,
                EntryID:  entryID,
            }
            
            s.logger.Printf("Added/Updated dynamic task: %s", id)
        }
    }
}
```

```go
package events

import (
    "context"
    "reflect"
    "sync"
    
    "github.com/jhseong7/gimbap"
)

// Event interface
type Event interface {
    GetEventType() string
}

// OrderCreatedEvent
type OrderCreatedEvent struct {
    OrderID    int64   `json:"orderId"`
    CustomerID int64   `json:"customerId"`
    Amount     float64 `json:"amount"`
    Timestamp  time.Time
}

func (e OrderCreatedEvent) GetEventType() string {
    return "order.created"
}

// EventBus handles event publishing and subscription
type EventBus struct {
    subscribers map[string][]EventHandler
    mu          sync.RWMutex
    
    // For async event handling
    eventQueue chan EventWrapper
    workers    int
    ctx        context.Context
    cancel     context.CancelFunc
    wg         sync.WaitGroup
}

type EventHandler func(context.Context, Event) error

type EventWrapper struct {
    Event   Event
    Context context.Context
}

func NewEventBus(workers int) *EventBus {
    ctx, cancel := context.WithCancel(context.Background())
    
    bus := &EventBus{
        subscribers: make(map[string][]EventHandler),
        eventQueue:  make(chan EventWrapper, 1000),
        workers:     workers,
        ctx:         ctx,
        cancel:      cancel,
    }
    
    // Start worker goroutines
    for i := 0; i < workers; i++ {
        bus.wg.Add(1)
        go bus.worker()
    }
    
    return bus
}

func (b *EventBus) worker() {
    defer b.wg.Done()
    
    for {
        select {
        case <-b.ctx.Done():
            return
        case wrapper := <-b.eventQueue:
            b.processEvent(wrapper)
        }
    }
}

func (b *EventBus) processEvent(wrapper EventWrapper) {
    eventType := wrapper.Event.GetEventType()
    
    b.mu.RLock()
    handlers := b.subscribers[eventType]
    b.mu.RUnlock()
    
    for _, handler := range handlers {
        if err := handler(wrapper.Context, wrapper.Event); err != nil {
            // Log error
            log.Printf("Error handling event %s: %v", eventType, err)
        }
    }
}

// Subscribe to events
func (b *EventBus) Subscribe(eventType string, handler EventHandler) {
    b.mu.Lock()
    defer b.mu.Unlock()
    
    b.subscribers[eventType] = append(b.subscribers[eventType], handler)
}

// Subscribe with condition
func (b *EventBus) SubscribeWithCondition(eventType string, condition func(Event) bool, handler EventHandler) {
    wrappedHandler := func(ctx context.Context, event Event) error {
        if condition(event) {
            return handler(ctx, event)
        }
        return nil
    }
    
    b.Subscribe(eventType, wrappedHandler)
}

// Publish event synchronously
func (b *EventBus) Publish(ctx context.Context, event Event) {
    b.processEvent(EventWrapper{Event: event, Context: ctx})
}

// Publish event asynchronously
func (b *EventBus) PublishAsync(ctx context.Context, event Event) {
    select {
    case b.eventQueue <- EventWrapper{Event: event, Context: ctx}:
    case <-ctx.Done():
    }
}

// Shutdown the event bus
func (b *EventBus) Shutdown() {
    b.cancel()
    close(b.eventQueue)
    b.wg.Wait()
}

var EventBusProvider = gimbap.DefineProvider(gimbap.ProviderOption{
    Name: "EventBus",
    Instantiator: func() *EventBus {
        return NewEventBus(5) // 5 worker goroutines
    },
})

// OrderService with events
type OrderService struct {
    orderRepo *repository.OrderRepository
    eventBus  *EventBus
}

func NewOrderService(orderRepo *repository.OrderRepository, eventBus *EventBus) *OrderService {
    return &OrderService{
        orderRepo: orderRepo,
        eventBus:  eventBus,
    }
}

func (s *OrderService) CreateOrder(ctx context.Context, request OrderRequest) (*Order, error) {
    // Begin transaction
    tx, err := s.orderRepo.BeginTx(ctx)
    if err != nil {
        return nil, err
    }
    defer tx.Rollback()
    
    // Create order
    order := &Order{
        CustomerID: request.CustomerID,
        Amount:     request.Amount,
        // ... other fields
    }
    
    if err := s.orderRepo.SaveTx(tx, order); err != nil {
        return nil, err
    }
    
    // Commit transaction
    if err := tx.Commit(); err != nil {
        return nil, err
    }
    
    // Publish event after commit
    event := OrderCreatedEvent{
        OrderID:    order.ID,
        CustomerID: order.CustomerID,
        Amount:     order.Amount,
        Timestamp:  time.Now(),
    }
    
    s.eventBus.PublishAsync(ctx, event)
    
    return order, nil
}

type OrderEventListener struct {
    microservice.IMicroService
    eventBus         *EventBus
    emailService     *EmailService
    inventoryService *InventoryService
}

func NewOrderEventListener(
    eventBus *EventBus,

    l.eventBus.Subscribe("order.created", l.handleOrderCreated)
    
    // Subscribe with condition (large orders)
    l.eventBus.SubscribeWithCondition("order.created",
        func(e Event) bool {
            if event, ok := e.(OrderCreatedEvent); ok {
                return event.Amount > 1000
            }
            return false
        },
        l.handleLargeOrder,
    )
}

    // Cleanup if needed
}

    orderEvent, ok := event.(OrderCreatedEvent)
    if !ok {
        return fmt.Errorf("invalid event type")
    }
    
    log.Printf("Order created: %d", orderEvent.OrderID)
    
    // Send confirmation email asynchronously
    go l.emailService.SendOrderConfirmation(orderEvent.CustomerID, orderEvent.OrderID)
    
    return nil
}

    orderEvent := event.(OrderCreatedEvent)

    return l.notifyManagement(orderEvent)

// Transactional event pattern
type TransactionalEventPublisher struct {
    eventBus *EventBus
    pending  []Event
    mu       sync.Mutex
}

func (p *TransactionalEventPublisher) PublishAfterCommit(event Event) {
    p.mu.Lock()
    defer p.mu.Unlock()
    
    p.pending = append(p.pending, event)
}

func (p *TransactionalEventPublisher) Commit(ctx context.Context) {
    p.mu.Lock()
    events := p.pending
    p.pending = nil
    p.mu.Unlock()
    
    for _, event := range events {
        p.eventBus.PublishAsync(ctx, event)
    }
}

func (p *TransactionalEventPublisher) Rollback() {
    p.mu.Lock()
    defer p.mu.Unlock()
    
    p.pending = nil
}
```

```go
// internal/queue/rabbitmq.go
package queue

import (
    "context"
    "encoding/json"
    "fmt"
    
    "github.com/streadway/amqp"
)

type RabbitMQClient struct {
    conn    *amqp.Connection
    channel *amqp.Channel
}

func NewRabbitMQClient(url string) (*RabbitMQClient, error) {
    conn, err := amqp.Dial(url)
    if err != nil {
        return nil, err
    }
    
    channel, err := conn.Channel()
    if err != nil {
        return nil, err
    }
    
    return &RabbitMQClient{
        conn:    conn,
        channel: channel,
    }, nil
}

func (c *RabbitMQClient) DeclareQueue(name string) error {
    _, err := c.channel.QueueDeclare(
        name,  // name
        true,  // durable
        false, // delete when unused
        false, // exclusive
        false, // no-wait
        nil,   // arguments
    )
    return err
}

func (c *RabbitMQClient) Publish(exchange, routingKey string, message interface{}) error {
    body, err := json.Marshal(message)
    if err != nil {
        return err
    }
    
    return c.channel.Publish(
        exchange,
        routingKey,
        false, // mandatory
        false, // immediate
        amqp.Publishing{
            ContentType: "application/json",
            Body:        body,
        },
    )
}

func (c *RabbitMQClient) Consume(queue string, handler func([]byte) error) error {
    msgs, err := c.channel.Consume(
        queue,
        "",    // consumer
        false, // auto-ack
        false, // exclusive
        false, // no-local
        false, // no-wait
        nil,   // args
    )
    if err != nil {
        return err
    }
    
    go func() {
        for msg := range msgs {
            if err := handler(msg.Body); err != nil {
                msg.Nack(false, true) // Requeue on error
            } else {
                msg.Ack(false)
            }
        }
    }()
    
    return nil
}

func (c *RabbitMQClient) Close() {
    c.channel.Close()
    c.conn.Close()
}

// Simple in-memory queue for development
type InMemoryQueue struct {
    queues map[string]chan Message
    mu     sync.RWMutex
}

func NewInMemoryQueue() *InMemoryQueue {
    return &InMemoryQueue{
        queues: make(map[string]chan Message),
    }
}

func (q *InMemoryQueue) CreateQueue(name string, size int) {
    q.mu.Lock()
    defer q.mu.Unlock()
    
    q.queues[name] = make(chan Message, size)
}

func (q *InMemoryQueue) Send(queueName string, message Message) error {
    q.mu.RLock()
    queue, exists := q.queues[queueName]
    q.mu.RUnlock()
    
    if !exists {
        return fmt.Errorf("queue %s not found", queueName)
    }
    
    select {
    case queue <- message:
        return nil
    default:
        return fmt.Errorf("queue %s is full", queueName)
    }
}

func (q *InMemoryQueue) Receive(queueName string) (<-chan Message, error) {
    q.mu.RLock()
    defer q.mu.RUnlock()
    
    queue, exists := q.queues[queueName]
    if !exists {
        return nil, fmt.Errorf("queue %s not found", queueName)
    }
    
    return queue, nil
}
```

## Best Practices

1. **Goroutine Management**
   - Always use context for cancellation
   - Limit concurrent goroutines with worker pools
   - Handle panics in goroutines

2. **Channel Usage**
   - Buffer channels appropriately
   - Always close channels when done
   - Use select for non-blocking operations

3. **Scheduling**
   - Use proper timezone handling
   - Implement distributed locking for clustered deployments
   - Monitor scheduled task execution

4. **Event Handling**
   - Make events immutable
   - Use async processing for heavy operations
   - Implement proper error handling and retries

5. **Message Queues**
   - Implement proper acknowledgment
   - Handle poison messages
   - Monitor queue depth and consumer lag
