# Step 4: Microservices and Background Tasks

This guide demonstrates how to convert NestJS background services, scheduled tasks, and event-driven patterns to GIMBAP's microservice architecture.

## Overview

We'll cover:
- Background services and lifecycle hooks
- Scheduled tasks (cron jobs)
- Event emitters and listeners
- Queue processing
- Health checks and monitoring

## Background Services

### NestJS Background Service

```typescript
import { Injectable, OnModuleInit, OnModuleDestroy } from '@nestjs/common';

@Injectable()
export class MetricsService implements OnModuleInit, OnModuleDestroy {

    console.log('MetricsService initialized');

  onModuleDestroy() {

    if (this.metricsInterval) {
      clearInterval(this.metricsInterval);
    }

    // Simulate metrics collection
    this.metricsInterval = setInterval(() => {
      this.collectMetrics();
    }, 5000);

    console.log('Current metrics:', this.metrics);
```

### GIMBAP Background Service

```go
// internal/services/metrics.go
package services

import (
    "fmt"
    "log"
    "math/rand"
    "sync"
    "time"
    
    "github.com/jhseong7/gimbap"
    "github.com/jhseong7/gimbap/microservice"
)

type Metrics struct {
    RequestCount      int `json:"requestCount"`
    ErrorCount        int `json:"errorCount"`
    ActiveConnections int `json:"activeConnections"`
}

type MetricsService struct {
    microservice.IMicroService
    mu             sync.RWMutex
    metrics        Metrics
    collectTicker  *time.Ticker
    reportTicker   *time.Ticker
    stopChan       chan bool
}

func NewMetricsService() *MetricsService {
    return &MetricsService{
        metrics:  Metrics{},
        stopChan: make(chan bool),
    }
}

// Start is called when the app starts
func (s *MetricsService) Start() {
    log.Println("MetricsService initialized")
    
    // Start collecting metrics every 5 seconds
    s.collectTicker = time.NewTicker(5 * time.Second)
    go s.collectMetricsLoop()
    
    // Report metrics every 10 seconds
    s.reportTicker = time.NewTicker(10 * time.Second)
    go s.reportMetricsLoop()
}

// Stop is called on app shutdown
func (s *MetricsService) Stop() {
    log.Println("MetricsService shutting down")
    
    // Stop tickers
    if s.collectTicker != nil {
        s.collectTicker.Stop()
    }
    if s.reportTicker != nil {
        s.reportTicker.Stop()
    }
    
    // Signal goroutines to stop
    close(s.stopChan)
}

func (s *MetricsService) collectMetricsLoop() {
    for {
        select {
        case <-s.collectTicker.C:
            s.collectMetrics()
        case <-s.stopChan:
            return
        }
    }
}

func (s *MetricsService) reportMetricsLoop() {
    for {
        select {
        case <-s.reportTicker.C:
            s.reportMetrics()
        case <-s.stopChan:
            return
        }
    }
}

func (s *MetricsService) collectMetrics() {
    s.mu.Lock()
    defer s.mu.Unlock()
    
    // Simulate metric collection
    s.metrics.RequestCount += rand.Intn(10)
    s.metrics.ActiveConnections = rand.Intn(50)
}

func (s *MetricsService) reportMetrics() {
    s.mu.RLock()
    defer s.mu.RUnlock()
    
    log.Printf("Current metrics: %+v", s.metrics)
}

func (s *MetricsService) IncrementRequestCount() {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.metrics.RequestCount++
}

func (s *MetricsService) IncrementErrorCount() {
    s.mu.Lock()
    defer s.mu.Unlock()
    s.metrics.ErrorCount++
}

func (s *MetricsService) GetMetrics() Metrics {
    s.mu.RLock()
    defer s.mu.RUnlock()
    
    // Return a copy
    return Metrics{
        RequestCount:      s.metrics.RequestCount,
        ErrorCount:        s.metrics.ErrorCount,
        ActiveConnections: s.metrics.ActiveConnections,
    }
}

// Register as microservice
var MetricsServiceProvider = microservice.DefineMicroService(
    microservice.MicroServiceProviderOption{
        Name:         "MetricsService",
        Instantiator: NewMetricsService,
    },
)
```

```go
// internal/events/events.go
package events

import (
    "sync"
)

type Event interface {
    GetName() string
}

type OrderCreatedEvent struct {
    OrderID string
    UserID  string
    Amount  float64
}

func (e OrderCreatedEvent) GetName() string {
    return "order.created"
}

// EventBus handles event publishing and subscription
type EventBus struct {
    mu        sync.RWMutex
    listeners map[string][]EventHandler
}

type EventHandler func(event Event)

func NewEventBus() *EventBus {
    return &EventBus{
        listeners: make(map[string][]EventHandler),
    }
}

func (eb *EventBus) Subscribe(eventName string, handler EventHandler) {
    eb.mu.Lock()
    defer eb.mu.Unlock()
    
    eb.listeners[eventName] = append(eb.listeners[eventName], handler)
}

func (eb *EventBus) Emit(event Event) {
    eb.mu.RLock()
    handlers := eb.listeners[event.GetName()]
    eb.mu.RUnlock()
    
    // Execute handlers asynchronously
    for _, handler := range handlers {
        go handler(event)
    }
}
```

### GIMBAP Queue Processing

```go
// internal/queues/queue.go
package queues

import (
    "encoding/json"
    "sync"
)

type Job struct {
    ID   string
    Type string
    Data json.RawMessage
}

type Queue struct {
    mu       sync.Mutex
    jobs     []Job
    handlers map[string]JobHandler
    stopChan chan bool
    wg       sync.WaitGroup
}

type JobHandler func(job Job) error

func NewQueue() *Queue {
    return &Queue{
        jobs:     make([]Job, 0),
        handlers: make(map[string]JobHandler),
        stopChan: make(chan bool),
    }
}

func (q *Queue) RegisterHandler(jobType string, handler JobHandler) {
    q.handlers[jobType] = handler
}

func (q *Queue) Add(jobType string, data interface{}) (string, error) {
    q.mu.Lock()
    defer q.mu.Unlock()
    
    dataBytes, err := json.Marshal(data)
    if err != nil {
        return "", err
    }
    
    job := Job{
        ID:   fmt.Sprintf("job-%d", time.Now().UnixNano()),
        Type: jobType,
        Data: dataBytes,
    }
    
    q.jobs = append(q.jobs, job)
    return job.ID, nil
}

func (q *Queue) Start(workers int) {
    for i := 0; i < workers; i++ {
        q.wg.Add(1)
        go q.worker()
    }
}

func (q *Queue) Stop() {
    close(q.stopChan)
    q.wg.Wait()
}

func (q *Queue) worker() {
    defer q.wg.Done()
    
    ticker := time.NewTicker(100 * time.Millisecond)
    defer ticker.Stop()
    
    for {
        select {
        case <-q.stopChan:
            return
        case <-ticker.C:
            q.processNextJob()
        }
    }
}

func (q *Queue) processNextJob() {
    q.mu.Lock()
    if len(q.jobs) == 0 {
        q.mu.Unlock()
        return
    }
    
    job := q.jobs[0]
    q.jobs = q.jobs[1:]
    q.mu.Unlock()
    
    handler, exists := q.handlers[job.Type]
    if !exists {
        log.Printf("No handler for job type: %s", job.Type)
        return
    }
    
    if err := handler(job); err != nil {
        log.Printf("Error processing job %s: %v", job.ID, err)
    }
}
```

### GIMBAP Health Check

```go
// internal/health/service.go
package health

import (
    "net/http"
    "os"
    "runtime"
    "syscall"
    "time"
    
    "github.com/jhseong7/gimbap"
)

type HealthStatus string

const (
    StatusUp   HealthStatus = "up"
    StatusDown HealthStatus = "down"
)

type HealthCheck struct {
    Name   string
    Status HealthStatus
    Data   map[string]interface{}
    Error  string
}

type HealthResult struct {
    Status string                 `json:"status"`
    Info   map[string]interface{} `json:"info"`
    Error  map[string]interface{} `json:"error,omitempty"`
    Details map[string]HealthCheck `json:"details"`
}

type HealthService struct {
    db         DatabaseChecker
    httpClient *http.Client
}

type DatabaseChecker interface {
    Ping() error
}

func NewHealthService(db DatabaseChecker) *HealthService {
    return &HealthService{
        db: db,
        httpClient: &http.Client{
            Timeout: 5 * time.Second,
        },
    }
}

func (s *HealthService) Check() HealthResult {
    checks := []HealthCheck{
        s.checkDatabase(),
        s.checkAPI(),
        s.checkMemory(),
        s.checkDisk(),
    }
    
    result := HealthResult{
        Status:  "ok",
        Info:    make(map[string]interface{}),
        Error:   make(map[string]interface{}),
        Details: make(map[string]HealthCheck),
    }
    
    for _, check := range checks {
        result.Details[check.Name] = check
        
        if check.Status == StatusDown {
            result.Status = "error"
            result.Error[check.Name] = map[string]interface{}{
                "status": check.Status,
                "error":  check.Error,
            }
        } else {
            result.Info[check.Name] = map[string]interface{}{
                "status": check.Status,
                "data":   check.Data,
            }
        }
    }
    
    return result
}

func (s *HealthService) checkDatabase() HealthCheck {
    check := HealthCheck{
        Name:   "database",
        Status: StatusUp,
        Data:   make(map[string]interface{}),
    }
    
    if err := s.db.Ping(); err != nil {
        check.Status = StatusDown
        check.Error = err.Error()
    }
    
    return check
}

func (s *HealthService) checkAPI() HealthCheck {
    check := HealthCheck{
        Name:   "external_api",
        Status: StatusUp,
        Data:   make(map[string]interface{}),
    }
    
    resp, err := s.httpClient.Get("https://api.example.com/health")
    if err != nil {
        check.Status = StatusDown
        check.Error = err.Error()
        return check
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        check.Status = StatusDown
        check.Error = fmt.Sprintf("API returned status %d", resp.StatusCode)
    }
    
    return check
}

func (s *HealthService) checkMemory() HealthCheck {
    var m runtime.MemStats
    runtime.ReadMemStats(&m)
    
    check := HealthCheck{
        Name:   "memory",
        Status: StatusUp,
        Data: map[string]interface{}{
            "heap_alloc_mb": m.Alloc / 1024 / 1024,
            "heap_sys_mb":   m.HeapSys / 1024 / 1024,
            "gc_count":      m.NumGC,
        },
    }
    
    // Check if heap allocation exceeds 150MB
    if m.Alloc > 150*1024*1024 {
        check.Status = StatusDown
        check.Error = "Memory usage exceeds threshold"
    }
    
    return check
}

func (s *HealthService) checkDisk() HealthCheck {
    check := HealthCheck{
        Name:   "disk",
        Status: StatusUp,
        Data:   make(map[string]interface{}),
    }
    
    var stat syscall.Statfs_t
    if err := syscall.Statfs("/", &stat); err != nil {
        check.Status = StatusDown
        check.Error = err.Error()
        return check
    }
    
    total := stat.Blocks * uint64(stat.Bsize)
    free := stat.Bavail * uint64(stat.Bsize)
    used := total - free
    usagePercent := float64(used) / float64(total)
    
    check.Data = map[string]interface{}{
        "total_gb":      total / 1024 / 1024 / 1024,
        "free_gb":       free / 1024 / 1024 / 1024,
        "used_percent":  usagePercent * 100,
    }
    
    // Check if disk usage exceeds 90%
    if usagePercent > 0.9 {
        check.Status = StatusDown
        check.Error = "Disk usage exceeds 90%"
    }
    
    return check
}

    Name:         "HealthService",
    Instantiator: NewHealthService,
```

## Registering Microservices in Main

```go
// cmd/main.go
func main() {
    app := gimbap.CreateApp(gimbap.AppOption{
        AppName:      "MyApp",
        AppModule:    app.AppModule,
        ServerEngine: gin_engine.NewGinHttpEngine(),
    })
    
    // Microservices are automatically started when included in modules
    // They will be gracefully stopped on app shutdown
    
    // Handle graceful shutdown
    sigChan := make(chan os.Signal, 1)
    signal.Notify(sigChan, syscall.SIGINT, syscall.SIGTERM)
    
    go func() {
        <-sigChan
        log.Println("Shutting down gracefully...")
        app.Stop() // This will call Stop() on all microservices
    }()
    
    app.Run()
}
```

## Best Practices

1. **Lifecycle Management**
   - Always implement proper Start() and Stop() methods
   - Clean up resources (tickers, channels, connections)
   - Use context for cancellation

2. **Concurrency**
   - Use sync.Mutex for shared state
   - Prefer channels for communication
   - Avoid goroutine leaks

3. **Error Handling**
   - Log errors appropriately
   - Implement retry logic for critical tasks
   - Use circuit breakers for external services

4. **Monitoring**
   - Implement comprehensive health checks
   - Export metrics for monitoring
   - Log important events

## Summary

Key differences when implementing microservices:
1. **Lifecycle Hooks**: Explicit Start/Stop methods vs decorators
2. **Scheduling**: External cron library vs built-in decorators
3. **Events**: Manual event bus vs built-in EventEmitter
4. **Queues**: Custom implementation vs Bull/BullMQ
